/* BEGIN DICOM PANEL CONTROLLER PHASE 5 */ function normalizeDicomPanelName(panelName) { const key = String(panelName || "").trim(); return key || "info"; } function getDicomPanel(panelName) { return document.getElementById("panel-" + normalizeDicomPanelName(panelName)); } function getDicomTab(panelName) { return document.getElementById("tab-" + normalizeDicomPanelName(panelName)); } function getRegisteredDicomPanel(panelName) { const key = normalizeDicomPanelName(panelName); }); document.querySelectorAll(".dicom-panel-section").forEach(function (section) { section.classList.toggle("active", section.id === "panel-" + key); }); panel.classList.add("dicom-panel-enter"); panel.addEventListener("animationend", function cleanupPanelEntry() { panel.classList.remove("dicom-panel-enter"); }, { once: true }); window.currentDicomPanel = key; try { sessionStorage.setItem("dicomLastPanel", key); } catch (error) { console.warn("Could not remember DICOM panel:", error); } updateDicomResultVisibility(key); updateResultToolbarContext(key); const registeredPanel = getRegisteredDicomPanel(key); if (registeredPanel && typeof registeredPanel.onShow === "function") { try { registeredPanel.onShow(panel); } catch (error) { console.warn("Panel onShow failed for", key, error); } } } function updateDicomResultVisibility(panelKey) { const module1 = document.getElementById("acrModule1CtNumberResultArea"); const module3 = document.getElementById("acrUniformityResultArea"); const module3Integrated = document.getElementById("automaticModule3OriginalResults"); const roi = document.getElementById("dicomRoiInfo"); const roiImages = document.getElementById("dicomRoiImages"); function hide(element) { if (element) { element.style.display = "none"; } } if (panelKey === "module1") { hide(module3); hide(module3Integrated); hide(roi); hide(roiImages); } else if (panelKey === "auto") { hide(module1); hide(roi); hide(roiImages); } else if (panelKey === "roi") { hide(module1); hide(module3); hide(module3Integrated); } } function setCurrentResultContext(context) { window.currentResultContext = String(context || ""); updateResultToolbarContext(window.currentDicomPanel || ""); } function updateResultToolbarContext(panelKey) { const toolbar = document.getElementById("resultContextActions"); const clearRoiButton = document.getElementById("clearRoiResultsButton"); const processedButton = document.getElementById("toggleProcessedImagesButton"); if (!toolbar || !clearRoiButton || !processedButton) { return; } const context = window.currentResultContext || ""; const viewingResults = panelKey === "results"; const viewingRoi = panelKey === "roi"; const hasRoiResults = window.UIResultDebug && window.UIResultDebug.roi !== undefined; const module3Processed = document.getElementById("automaticModule3ProcessedImages"); const hasModule3Processed = Boolean( module3Processed && module3Processed.innerHTML.trim() ); const showRoiClear = context === "roi" && hasRoiResults && (viewingRoi || viewingResults); "" : "none"; return; } } const originalImage = document.getElementById("dicomOriginalImage"); const host = originalImage ? originalImage.closest(".simple-info, .visible-info-box, .side-card, .card, section, article, div") : null; if (host) { host.style.display = host.style.display === "none" ? "" : "none"; return; } console.warn("No processed images host found to toggle."); } function verifyDicomPanelMap() { const tabs = Array.from(document.querySelectorAll(".dicom-tab[id^='tab-']")).map(function (tab) { return tab.id.replace(/^tab-/, ""); }); const panels = Array.from(document.querySelectorAll(".dicom-panel-section[id^='panel-']")).map(function (panel) { return panel.id.replace(/^panel-/, ""); }); const panelSet = new Set(panels); const tabSet = new Set(tabs); const missingPanels = tabs.filter(function (key) { return !panelSet.has(key); }); const missingTabs = panels.filter(function (key) { return !tabSet.has(key); }); if (missingPanels.length || missingTabs.length) { console.warn("DICOM tab/panel mismatch", { tabsWithoutPanels: missingPanels, panelsWithoutTabs: missingTabs }); } return { tabs: tabs, panels: panels, tabsWithoutPanels: missingPanels, panelsWithoutTabs: missingTabs }; } document.addEventListener("DOMContentLoaded", function () { const map = verifyDicomPanelMap(); const activeTab = document.querySelector(".dicom-tab.active[id^='tab-']"); const activePanel = document.querySelector(".dicom-panel-section.active[id^='panel-']"); if (activeTab && activePanel) { const tabKey = activeTab.id.replace(/^tab-/, ""); const panelKey = activePanel.id.replace(/^panel-/, ""); if (tabKey === panelKey) { return; } } if (activeTab) { switchDicomPanel(activeTab.id.replace(/^tab-/, "")); return; } if (map.panels.indexOf("info") !== -1) { switchDicomPanel("info"); } }); /* END DICOM PANEL CONTROLLER PHASE 5 */
